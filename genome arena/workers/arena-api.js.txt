const ORIGIN = 'https://bread11yt.github.io';
const cors = {
  'Access-Control-Allow-Origin': ORIGIN,
  'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',
  'Access-Control-Allow-Headers': 'Content-Type',
};
const json = (data, status = 200) =>
  new Response(JSON.stringify(data), {
    status,
    headers: { ...cors, 'Content-Type': 'application/json' },
  });

const enc = new TextEncoder();
async function hmac(key, data) {
  const k = await crypto.subtle.importKey('raw', key, { name: 'HMAC', hash: 'SHA-256' }, false, ['sign']);
  return new Uint8Array(await crypto.subtle.sign('HMAC', k, enc.encode(data)));
}
const hex = b => [...b].map(x => x.toString(16).padStart(2, '0')).join('');

// Проверяем, что данные действительно пришли из Telegram
async function verify(initData, token) {
  const p = new URLSearchParams(initData);
  const hash = p.get('hash');
  if (!hash) return null;
  p.delete('hash');
  const str = [...p.entries()]
    .sort(([a], [b]) => (a < b ? -1 : 1))
    .map(([k, v]) => `${k}=${v}`)
    .join('\n');
  const secret = await hmac(enc.encode('WebAppData'), token);
  if (hex(await hmac(secret, str)) !== hash) return null;
  const age = Date.now() / 1000 - Number(p.get('auth_date'));
  if (!(age < 7 * 86400)) return null;
  try { return JSON.parse(p.get('user')); } catch { return null; }
}

export default {
  async fetch(req, env) {
    if (req.method === 'OPTIONS') return new Response(null, { headers: cors });
    const url = new URL(req.url);

    // Отправка рекорда
    if (req.method === 'POST' && url.pathname === '/score') {
      let body;
      try { body = await req.json(); } catch { return json({ error: 'bad json' }, 400); }

      const user = await verify(String(body.initData || ''), env.BOT_TOKEN);
      if (!user) return json({ error: 'unauthorized' }, 401);

      let best = Math.floor(Number(body.best));
      if (!(best >= 1 && best <= 100000)) return json({ error: 'bad score' }, 400);

      const id = String(user.id);
      const name = ([user.first_name, user.last_name].filter(Boolean).join(' ') || user.username || 'Игрок').slice(0, 40);

      const row = await env.DB.prepare('SELECT best FROM scores WHERE user_id=?').bind(id).first();
      const old = row ? row.best : 0;

      // Простая защита от накрутки: рост за один запрос ограничен
      best = Math.min(best, row ? old + 3 : 50);

      if (best > old) {
        await env.DB.prepare(
          'INSERT INTO scores (user_id,name,best,updated_at) VALUES (?,?,?,?) ' +
          'ON CONFLICT(user_id) DO UPDATE SET best=excluded.best, name=excluded.name, updated_at=excluded.updated_at'
        ).bind(id, name, best, Date.now()).run();
      } else {
        await env.DB.prepare('UPDATE scores SET name=? WHERE user_id=?').bind(name, id).run();
      }

      const final = Math.max(old, best);
      const r = await env.DB.prepare('SELECT COUNT(*)+1 AS r FROM scores WHERE best > ?').bind(final).first();
      return json({ ok: true, best: final, rank: r.r });
    }

    // Топ-20
    if (req.method === 'GET' && url.pathname === '/top') {
      const { results } = await env.DB.prepare(
        'SELECT name, best FROM scores ORDER BY best DESC, updated_at ASC LIMIT 20'
      ).all();
      return json({ top: results });
    }

    return json({ error: 'not found' }, 404);
  },
};
